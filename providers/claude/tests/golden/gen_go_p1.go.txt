// p1Spec is the AgentSpec projection the Rust golden tests rebuild as a
// TurnRequest (inject = a steerable turn).
type p1Spec struct {
	WorkspaceID        string            `json:"workspaceId"`
	AgentID            string            `json:"agentId"`
	Role               string            `json:"role"`
	Prompt             string            `json:"prompt"`
	Model              string            `json:"model"`
	Flags              map[string]string `json:"flags"`
	SystemPrompt       string            `json:"systemPrompt"`
	AppendSystemPrompt string            `json:"appendSystemPrompt"`
	TurnAppendPrompt   string            `json:"turnAppendPrompt"`
	RefreshStanding    bool              `json:"refreshStanding"`
	AllowedTools       []string          `json:"allowedTools"`
	Tools              *string           `json:"tools"`
	PermissionMode     string            `json:"permissionMode"`
	MCPURL             string            `json:"mcpUrl"`
	Playwright         bool              `json:"playwright"`
	UserMCP            []p1MCP           `json:"userMcp"`
	ResumeSession      string            `json:"resumeSession"`
	ForkSession        bool              `json:"forkSession"`
	PromptOnStdin      bool              `json:"promptOnStdin"`
	IncludePartials    bool              `json:"includePartials"`
	SkipThinking       bool              `json:"skipThinking"`
	MaxTurns           int               `json:"maxTurns"`
	Images             []p1Image         `json:"images"`
	Inject             bool              `json:"inject"`
}

func (s p1Spec) agentSpec() AgentSpec {
	spec := AgentSpec{
		WorkspaceID: s.WorkspaceID, AgentID: s.AgentID, Role: domain.AgentRole(s.Role),
		Prompt: s.Prompt, Model: s.Model, Flags: s.Flags, SystemPrompt: s.SystemPrompt,
		AppendSystemPrompt: s.AppendSystemPrompt, TurnAppendPrompt: s.TurnAppendPrompt,
		RefreshStandingInstructions: s.RefreshStanding, AllowedTools: s.AllowedTools, Tools: s.Tools,
		PermissionMode: s.PermissionMode, MCPURL: s.MCPURL, Playwright: s.Playwright,
		ResumeSession: s.ResumeSession, ForkSession: s.ForkSession, PromptOnStdin: s.PromptOnStdin,
		IncludePartials: s.IncludePartials, SkipThinking: s.SkipThinking, MaxTurns: s.MaxTurns,
	}
	for _, u := range s.UserMCP {
		spec.UserMCP = append(spec.UserMCP, MCPAttach{Name: u.Name, URL: u.URL, Headers: u.Headers, Command: u.Command, Args: u.Args, Env: u.Env})
	}
	for _, i := range s.Images {
		spec.Images = append(spec.Images, UserImage{ID: i.ID, Name: i.Name, Mime: i.Mime, Data: i.Data, Path: i.Path})
	}
	if s.Inject {
		spec.Inject = make(chan string)
	}
	return spec
}

func p1Event(ev ParsedEvent) map[string]any {
	m := map[string]any{"kind": string(ev.Kind)}
	set := func(k string, v any, keep bool) {
		if keep {
			m[k] = v
		}
	}
	set("tool", ev.Tool, ev.Tool != "")
	set("toolInput", ev.ToolInput, ev.ToolInput != "")
	set("toolInputRaw", string(ev.ToolInputRaw), len(ev.ToolInputRaw) > 0)
	set("toolUseId", ev.ToolUseID, ev.ToolUseID != "")
	set("text", ev.Text, ev.Text != "")
	set("isError", ev.IsError, ev.IsError)
	if ev.Result != nil {
		m["result"] = map[string]any{
			"in": ev.Result.InputTokens, "out": ev.Result.OutputTokens,
			"cr": ev.Result.CacheReadTokens, "cc": ev.Result.CacheCreationTokens, "cw": ev.Result.ContextWindow,
		}
	}
	return m
}

		if st.window > 0 {
			window = st.window
		}
		if !baselineSet && !tokenUsageEmpty(st.total) {
			baseline = subTokenUsage(st.total, st.last)
			baselineSet = true
		}
		if baselineSet {
			usageTurn = subTokenUsage(st.total, baseline)
		} else {
			usageTurn = st.last
		}
		var ev, turn Result
		applyCodexMeterUsage(&ev, usageLast, window)
		stampCodexTurnUsage(&turn, usageLast, usageTurn, window)
		lastJSON, _ := json.Marshal(st.last)
		var lastMap map[string]any
		_ = json.Unmarshal(lastJSON, &lastMap)
		meter := func(r Result) map[string]any {
			return map[string]any{"in": r.InputTokens, "out": r.OutputTokens, "cr": r.CacheReadTokens, "cc": r.CacheCreationTokens, "cw": r.ContextWindow}
		}
		usageRows = append(usageRows, usageRow{Params: p, Last: lastMap, Event: meter(ev), Turn: meter(turn)})
	}

func slicesIndex(s []string, v string) int {
