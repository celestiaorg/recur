import (
	"encoding/json"
	"os"
	"testing"
)

func TestZZGolden(t *testing.T) {
	out := map[string]map[string]bool{}
	for _, pat := range []string{".env*", "apps/.env", "apps/**/.env.local", "secrets/", "*.local", "config/*.json", "**", "a+b(c).txt", "?x", "a//", "dir/sub/"} {
		m := map[string]bool{}
		for _, p := range []string{".env", "apps/.env", "apps/web/.env.local", "apps/.env.local", "secrets/a", "x/secrets/b", "c.local", "d/c.local", "config/a.json", "config/b/a.json", "a+b(c).txt", "zx", "z/zx", "README", "a/x", "dir/sub/f", "q/dir/sub/f", "/abs/.env", "\\win\\.env"} {
			m[p] = Match(p, []string{pat})
		}
		out[pat] = m
	}
	parsed := map[string][]string{}
	for _, text := range []string{"\n.env*\n# c\n  a\\b  \n//x\n!neg\n/\n", "", "one"} {
		parsed[text] = ParsePatterns(text)
	}
	b, _ := json.MarshalIndent(map[string]any{"match": out, "parse": parsed}, "", "  ")
	os.WriteFile("../../rust/crates/code/tests/golden/go_copyfiles.json", b, 0o644)
}
