//	P3_USAGE_OUT=$PWD/rust/crates/usage/tests/golden/go_usage.json \

func p3jwt(claims string) string {
	return "eyJhbGciOiJub25lIn0." + base64.RawURLEncoding.EncodeToString([]byte(claims)) + "."
}
