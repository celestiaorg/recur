type recHost struct{ calls []map[string]any }

func (h *recHost) rec(m map[string]any) { h.calls = append(h.calls, m) }

func nz(s []string) []string {
	if s == nil {
		return []string{}
	}
	return s
}

func opts(o []EscalationOption) []any {
	out := []any{}
	for _, x := range o {
		out = append(out, map[string]any{"text": x.Text, "tag": x.Tag, "desc": x.Desc})
	}
	return out
}

func (h *recHost) ReportFinding(ws, a string, f Finding) domain.Finding {
	h.rec(map[string]any{"method": "ReportFinding", "ws": ws, "agent": a, "file": f.File, "line": f.Line, "endLine": f.EndLine, "severity": string(f.Severity), "message": f.Message, "summary": f.Summary})
	return domain.Finding{ID: "uuid-1", ShortID: "find_1", WorkspaceID: ws, File: f.File, Line: f.Line, EndLine: f.EndLine, Severity: f.Severity, Message: f.Message, Summary: f.Summary, Status: "open"}
}
func (h *recHost) Done(ws, a, summary, notes string) error {
	h.rec(map[string]any{"method": "Done", "ws": ws, "agent": a, "summary": summary, "notes": notes})
	if summary == "blocked" {
		return errors.New("2 findings are still open")
	}
	return nil
}
func (h *recHost) ListFindings(ws, a, status string) ([]domain.Finding, error) {
	h.rec(map[string]any{"method": "ListFindings", "ws": ws, "agent": a, "status": status})
	if status == "invalid" {
		return nil, errors.New("store offline")
	}
	if status == "ignored" {
		return nil, nil
	}
	return []domain.Finding{
		{ID: "uuid-a", ShortID: "find_a", File: "a.go", Line: 3, EndLine: 9, Severity: domain.SevHigh, Message: "x < y && z", Summary: "sum", Status: "open", StatusReason: "", UpdatedBy: "coder",
			Notes: []domain.FindingNote{{ID: "n1", Author: "human", Body: "look <here>", CreatedAt: "2026-01-02T03:04:05Z"}}},
		{ID: "uuid-b", File: "b.go", Severity: domain.SevLow, Message: "m", Status: "resolved", StatusReason: "fixed"},
	}, nil
}
func (h *recHost) UpdateFinding(ws, a, id, status, reason string) (domain.Finding, error) {
	h.rec(map[string]any{"method": "UpdateFinding", "ws": ws, "agent": a, "id": id, "status": status, "reason": reason})
	if id == "missing" {
		return domain.Finding{}, fmt.Errorf("finding %q not found", id)
	}
	f := domain.Finding{ID: "uuid-" + id, ShortID: id, File: "c.go", Line: 7, Severity: domain.SevMedium, Message: "msg", Status: status, StatusReason: reason, UpdatedBy: "coder"}
	if id == "noted" {
		f.Notes = []domain.FindingNote{{ID: "n2", Author: "human", Body: "b", CreatedAt: "t"}}
	}
	return f, nil
}

func (h *recHost) RequestPermission(_ context.Context, ws, a string, ask PermissionAsk) (PermissionVerdict, error) {
	h.rec(map[string]any{"method": "RequestPermission", "ws": ws, "agent": a, "tool": ask.Tool, "input": string(ask.Input), "toolUseId": ask.ToolUseID})
	switch ask.Tool {
	case "Deny":
		return PermissionVerdict{Reason: "  not here  "}, nil
	case "Blank":
		return PermissionVerdict{}, nil
	case "Err":
		return PermissionVerdict{}, errors.New("broker down")
	}
	return PermissionVerdict{Allow: true}, nil
}

func runHost(t *testing.T, tool Tool, ws, agent, chat, args string, noHost bool) hostCase {
	h := &recHost{}
	c := Call{WorkspaceID: ws, AgentCode: agent, ChatCode: chat, Args: json.RawMessage(args)}
	if args == "" {
		c.Args = nil
	}
	if !noHost {
		c.Host = h
	}
	res, err := tool.Execute(c)
	out := hostCase{Tool: tool.Name(), Agent: agent, WS: ws, Chat: chat, Args: args, NoHost: noHost, Result: res, Calls: h.calls}
	if err != nil {
		out.Error = err.Error()
		out.IsError = true
	}
	if out.Calls == nil {
		out.Calls = []map[string]any{}
	}
	return out
}

func writeGolden(t *testing.T, name string, v any) {
	dir := filepath.Join("..", "..", "rust", "crates", "agent-tools", "tests", "golden")
	if err := os.MkdirAll(dir, 0o755); err != nil {
		t.Fatal(err)
	}
	b, err := json.MarshalIndent(v, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(dir, name), append(b, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}

type catalogEntry struct {
	Name        string `json:"name"`
	Description string `json:"description"`
	Schema      string `json:"schema"`
	Halts       bool   `json:"halts"`
}

type codingCase struct {
	Tool    string `json:"tool"`
	Args    string `json:"args"`
	Bypass  bool   `json:"bypass"`
	Session string `json:"session"`
	Result  string `json:"result"`
	Error   string `json:"error"`
	IsError bool   `json:"isError"`
}

type denyAll struct{}

func (denyAll) RequestPermission(_ context.Context, _, _, tool string, input json.RawMessage) (bool, string, error) {
	switch string(input) {
	case `{"command":"echo blank"}`:
		return false, "", nil
	case `{"command":"echo err"}`:
		return false, "", errors.New("broker down")
	case `{"command":"echo ok"}`:
		return true, "", nil
	}
	return false, "no " + tool, nil
}

func normWT(s, wt string) string {
	if eval, err := filepath.EvalSymlinks(wt); err == nil {
		s = strings.ReplaceAll(s, eval, "$WT")
	}
	return strings.ReplaceAll(s, wt, "$WT")
}
