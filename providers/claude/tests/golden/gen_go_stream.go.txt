// Generator for go_stream.json. To regenerate, copy into internal/engine as
// zz_rustgolden_test.go and run from the repo root:
//
//	RECUR_RUST_GOLDEN=$PWD/providers/claude/tests/golden/go_stream.json \
//	  go test ./internal/engine -run TestDumpRustStreamGolden -count=1
//
// then delete the copy. Each row is {line, error, events}; events are
// normalized per kind to the fields recur_agent::AgentEvent carries, and the
// generator fails if the Go parser sets any field outside that projection.
package engine

import (
	"encoding/json"
	"os"
	"reflect"
	"strings"
	"testing"
)

func TestDumpRustStreamGolden(t *testing.T) {
	path := os.Getenv("RECUR_RUST_GOLDEN")
	if path == "" {
		t.Skip()
	}
	type row struct {
		Line   string           `json:"line"`
		Error  bool             `json:"error"`
		Events []map[string]any `json:"events"`
	}
	var rows []row
	for _, line := range streamCorpus() {
		evs, err := parseStreamLine([]byte(line))
		r := row{Line: line, Error: err != nil, Events: []map[string]any{}}
		for _, ev := range evs {
			r.Events = append(r.Events, normalizeEvent(t, ev))
		}
		rows = append(rows, r)
	}
	b, err := json.MarshalIndent(rows, "", " ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(path, append(b, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}

func normalizeEvent(t *testing.T, ev ParsedEvent) map[string]any {
	t.Helper()
	used := map[string]bool{"Kind": true}
	m := map[string]any{"kind": string(ev.Kind)}
	take := func(field, key string, v any) {
		used[field] = true
		m[key] = v
	}
	res := func(r *Result) map[string]any {
		used["Result"] = true
		if r == nil {
			return nil
		}
		return map[string]any{
			"sessionId": r.SessionID, "subtype": r.Subtype, "isError": r.IsError, "numTurns": r.NumTurns,
			"durationMs": r.DurationMS, "text": r.Text, "in": r.InputTokens, "out": r.OutputTokens,
			"cc": r.CacheCreationTokens, "cr": r.CacheReadTokens, "cw": r.ContextWindow, "stderr": r.Stderr,
		}
	}
	switch ev.Kind {
	case KindInit:
		take("SessionID", "sessionId", ev.SessionID)
		take("Model", "model", ev.Model)
		mcp := [][]string{}
		for _, s := range ev.MCPServers {
			mcp = append(mcp, []string{s.Name, s.Status})
		}
		take("MCPServers", "mcp", mcp)
	case KindText, KindThinking:
		take("Text", "text", ev.Text)
		take("Signature", "signature", ev.Signature)
		take("UUID", "uuid", ev.UUID)
		take("ParentToolUseID", "parent", ev.ParentToolUseID)
	case KindTextDelta, KindThinkingDelta:
		take("Text", "text", ev.Text)
		take("ParentToolUseID", "parent", ev.ParentToolUseID)
	case KindThinkingTokens:
		take("Tokens", "tokens", ev.Tokens)
		take("UUID", "uuid", ev.UUID)
		take("ParentToolUseID", "parent", ev.ParentToolUseID)
	case KindStep:
		take("ToolUseID", "id", ev.ToolUseID)
		take("Tool", "name", ev.Tool)
		take("ToolInput", "summary", ev.ToolInput)
		take("ToolInputRaw", "input", string(ev.ToolInputRaw))
		take("ParentToolUseID", "parent", ev.ParentToolUseID)
		take("UUID", "uuid", ev.UUID)
	case KindToolResult:
		take("ToolUseID", "id", ev.ToolUseID)
		take("Tool", "name", ev.Tool)
		take("ToolInputRaw", "input", string(ev.ToolInputRaw))
		take("Text", "text", ev.Text)
		take("IsError", "isError", ev.IsError)
		take("Truncated", "truncated", ev.Truncated)
		take("ParentToolUseID", "parent", ev.ParentToolUseID)
		take("UUID", "uuid", ev.UUID)
	case KindUsage:
		// The frame uuid is stamped on usage too; nothing reads it there.
		used["UUID"] = true
		m["usage"] = res(ev.Result)
	case KindNotice:
		take("Text", "text", ev.Text)
	case KindCompact:
		take("CompactStatus", "status", ev.CompactStatus)
		take("CompactTrigger", "trigger", ev.CompactTrigger)
		take("Text", "summary", ev.Text)
		take("Tokens", "preTokens", ev.Tokens)
		take("UUID", "uuid", ev.UUID)
		take("ParentToolUseID", "parent", ev.ParentToolUseID)
	case KindFallback:
		take("FromModel", "from", ev.FromModel)
		take("ToModel", "to", ev.ToModel)
		r := ev.RetractedUUIDs
		if r == nil {
			r = []string{}
		}
		take("RetractedUUIDs", "retracted", r)
		take("UUID", "uuid", ev.UUID)
	case KindResult:
		if ev.Result == nil || ev.Text != ev.Result.Text || ev.IsError != ev.Result.IsError {
			t.Fatalf("result headline drift: %+v", ev)
		}
		used["Text"], used["IsError"] = true, true
		m["result"] = res(ev.Result)
	default:
		t.Fatalf("kind %s not projected", ev.Kind)
	}
	v := reflect.ValueOf(ev)
	for i := 0; i < v.NumField(); i++ {
		f := v.Type().Field(i)
		if !used[f.Name] && !v.Field(i).IsZero() {
			t.Fatalf("%s event sets %s outside the projection: %+v", ev.Kind, f.Name, ev)
		}
	}
	return m
}
