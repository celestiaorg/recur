package app

// Generates providers/compatible/tests/golden/go_catalog.json (slice P3):
// the compatible branch of the model catalogue and the flag presets.
//
//	P3_CATALOG_OUT=$PWD/providers/compatible/tests/golden/go_catalog.json \
//	  go test ./internal/app -run TestP3CatalogGolden -count=1

import (
	"encoding/json"
	"os"
	"testing"

	"recur/internal/domain"
)

func TestP3CatalogGolden(t *testing.T) {
	out := os.Getenv("P3_CATALOG_OUT")
	if out == "" {
		t.Skip("P3_CATALOG_OUT not set")
	}
	providers := []domain.CompatibleProvider{
		{
			ID: "openai", Name: "OpenAI", BaseURL: "https://api.openai.com",
			Models: []domain.CompatibleModel{
				{ID: "gpt-5", Label: "GPT-5", ContextWindow: 400000, Flags: domain.CompatibleFlagPreset("openai")},
				{ID: "", Label: "skipped"},
				{ID: "o4-mini"},
				{ID: "openai/gpt-4o", Label: "4o", Flags: []domain.CompatibleFlagDim{{ID: "x", Label: "X", Wire: "temperature"}}},
			},
		},
		{ID: "groq", Name: "Groq", BaseURL: "https://api.groq.com/openai", Models: []domain.CompatibleModel{{ID: "llama-3.1-70b", Label: "Llama"}}},
		{ID: "local", BaseURL: "http://127.0.0.1:1234", Models: []domain.CompatibleModel{{ID: "qwen"}}},
		{ID: "nobase", Name: "No base", Models: []domain.CompatibleModel{{ID: "m"}}},
		{ID: "nomodels", Name: "No models", BaseURL: "https://x.example"},
	}
	var models []map[string]any
	for _, p := range providers {
		models = append(models, map[string]any{"provider": p, "models": compatibleCatalogModels(p)})
	}
	all := domain.Settings{
		DisabledProviders:   []string{domain.ProviderClaudeCode, domain.ProviderCursor, domain.ProviderCodex, "groq"},
		CompatibleProviders: providers,
	}
	presets := map[string]any{}
	for _, name := range []string{"openai", " OpenAI ", "openai-reasoning", "reasoning", "effort", "custom", ""} {
		presets[name] = (&SettingsService{}).CompatibleFlagPreset(name)
	}
	raw, err := json.MarshalIndent(map[string]any{
		"models":   models,
		"settings": all,
		"groups":   buildModelCatalog(all, nil, nil),
		"presets":  presets,
	}, "", " ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(out, append(raw, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
