func TestDumpForgeGolden(t *testing.T) {
	path := os.Getenv("RECUR_FORGE_GOLDEN")
	if path == "" {
		t.Skip()
	}
	now := time.Date(2026, 8, 21, 12, 0, 45, 0, time.UTC)
	out := map[string]any{"now": now.Format(time.RFC3339)}

	out["queries"] = map[string]string{
		"checks":           githubPRChecksGraphQL,
		"snapshot":         githubPRSnapshotGraphQL,
		"snapshotNoRollup": githubPRSnapshotGraphQLQuery(false),
	}

	snap, err := parseGitHubPRSnapshotGraphQL(goldenSnapshot, now)
	if err != nil {
		t.Fatal(err)
	}
	snap = finishGitHubPRSnapshot(snap, flattenJSONArrayPages(goldenTimeline), now)
	viewCommits := commitsFromGitHubView(snap.viewCommits, now)
	stored := []domain.Commit{{Hash: "abc1230", Author: "local"}, {SHA: "def4560000000000", Hash: "def4560", Author: "Josh Dechant", AvatarURL: "x"}}
	enriched := enrichGitHubCommits(stored, snap.commitAuthors, snap.commitMeta)
	applyGHCommitAvatars(enriched, snap.authorAvatars)
	out["snapshot"] = map[string]any{
		"input":               goldenSnapshot,
		"timeline":            goldenTimeline,
		"title":               snap.Title,
		"state":               snap.State,
		"reviewDecision":      snap.ReviewDecision,
		"viewerLogin":         snap.ViewerLogin,
		"viewerCanWrite":      snap.ViewerCanWrite,
		"allowedMergeMethods": snap.AllowedMergeMethods,
		"nodeID":              snap.NodeID,
		"mergeQueued":         snap.MergeQueued,
		"comments":            snap.Comments,
		"viewCommits":         viewCommits,
		"enrichedCommits":     enriched,
		"participants":        snapshotParticipants(snap, snap.Comments),
		"reviewers":           snapshotReviewers(snap, snap.Comments),
		"authorAvatars":       snap.authorAvatars,
	}

	cg, okG, errG := parseGitHubPRChecksGraphQLAt(goldenChecksGraphQL, now)
	cc, okC, errC := parseGitHubPRChecksAt(goldenCLIChecks, now)
	out["checks"] = map[string]any{
		"graphqlInput": goldenChecksGraphQL, "graphql": cg, "graphqlOK": okG, "graphqlErr": errG != nil,
		"cliInput": goldenCLIChecks, "cli": cc, "cliOK": okC, "cliErr": errC != nil,
	}

	rows, err := parseGitHubPRList(goldenList)
	if err != nil {
		t.Fatal(err)
	}
	var listed []map[string]any
	for _, r := range rows {
		checks, ok, has := listedChecks(r)
		state, group := listedStateGroup(r, "me")
		listed = append(listed, map[string]any{
			"checks": checks, "checksOK": ok, "hasChecks": has,
			"participants": listedParticipants(r), "reviewers": listedReviewers(r),
			"reasons": listedAttentionReasons(r, "me"), "reviewRequested": listedReviewRequested(r, "me"),
			"state": state, "group": group, "behind": listedBehind(r),
			"mergeable": listedMergeField("CONFLICTING", r.Mergeable), "createdAt": listedCreatedAt(r),
		})
	}
	out["list"] = map[string]any{"input": goldenList, "rows": listed}

	pr := domain.PR{Number: 12, Source: domain.SourceGitHub, Title: "Add OAuth\nsecond", Commits: []domain.Commit{{Message: "b"}, {Message: "a\nx"}},
		Comments: []domain.PRComment{{Kind: "description", BodyMD: " Why "}}}
	var messages []mergeCommitText
	for _, m := range []domain.MergeMethod{domain.MergeMethodSquash, domain.MergeMethodMerge, domain.MergeMethodRebase} {
		for _, s := range []domain.SquashCommitMessage{domain.SquashCommitTitle, domain.SquashCommitTitleAndCommits, domain.SquashCommitTitleAndDescription} {
			for _, mc := range []domain.MergeCommitMessage{domain.MergeCommitTitle, domain.MergeCommitTitleAndDescription} {
				messages = append(messages, formatPRMergeMessage(pr, m, s, mc))
			}
		}
	}
	out["mergeMessages"] = messages

	raw, err := json.MarshalIndent(out, "", " ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(path, raw, 0o644); err != nil {
		t.Fatal(err)
	}
}
