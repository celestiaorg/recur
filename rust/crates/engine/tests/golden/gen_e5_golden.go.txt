func TestE5GoldenDump(t *testing.T) {

		"prs":             e5PRs(),
		"squash":          e5Squash(),
		"merge":           e5Merge(),
		"local_merge":     e5LocalMerge(),

		"distill":         e5Distill(),
		"summary":         e5Summary(),
		"files":           e5Files,
		"hunks":           e5Hunks(),
		"pending_loc":     e5Pending(),
		"small":           e5Small(),
		"thread":          e5Thread,
		"thread_keys":     e5ThreadKeys(),
		"commit_log":      e5CommitLog(),
	}
	b, err := json.MarshalIndent(out, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile("../../rust/crates/engine/tests/golden/go_e5.json", append(b, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
