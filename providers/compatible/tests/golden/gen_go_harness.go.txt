func p3Raw(t *testing.T, v any) json.RawMessage {
	t.Helper()
	b, err := json.Marshal(v)
	if err != nil {
		t.Fatal(err)
	}
	return b
}

func p3Event(ev responses.Event) map[string]any {
	m := map[string]any{
		"type":         ev.Type,
		"responseId":   ev.ResponseID,
		"delta":        ev.Delta,
		"itemId":       ev.ItemID,
		"status":       ev.Status,
		"errorMessage": ev.ErrorMessage,
		"outputText":   ev.OutputText,
		"call":         nil,
		"usage":        nil,
	}
	if ev.Call != nil {
		m["call"] = map[string]any{"id": ev.Call.ID, "callId": ev.Call.CallID, "name": ev.Call.Name, "arguments": ev.Call.Arguments}
	}
	if ev.Usage != nil {
		m["usage"] = map[string]any{"inputTokens": ev.Usage.InputTokens, "outputTokens": ev.Usage.OutputTokens, "cacheReadTokens": ev.Usage.CacheReadTokens, "totalTokens": ev.Usage.TotalTokens}
	}
	return m
}

// p3RunGolden drives one full HarnessRunner.Run against scripted responses and
// records the exact request bodies and emitted events.
func p3RunGolden(t *testing.T) map[string]any {
	round1 := strings.Join([]string{
		`data: {"type":"response.created","response":{"id":"resp-1","status":"in_progress"}}`,
		``,
		`data: {"type":"response.reasoning_summary_text.delta","item_id":"rs_1","delta":"Thinking "}`,
		``,
		`data: {"type":"response.reasoning_summary_text.delta","item_id":"rs_1","delta":"hard"}`,
		``,
		`data: {"type":"response.reasoning_summary_text.done","item_id":"rs_1","text":"Thinking hard"}`,
		``,
		`data: {"type":"response.output_text.delta","item_id":"msg_1","delta":"Checking."}`,
		``,
		`data: {"type":"response.output_item.added","item":{"type":"function_call","id":"fc_1","call_id":"call_read","name":"Read","arguments":""}}`,
		``,
		`data: {"type":"response.function_call_arguments.delta","item_id":"fc_1","delta":"{bad"}`,
		``,
		`data: {"type":"response.output_item.done","item":{"type":"function_call","id":"fc_2","call_id":"call_todo","name":"TodoWrite","arguments":"{\"todos\":[{\"id\":\"1\",\"content\":\"Look\",\"status\":\"in_progress\"}]}"}}`,
		``,
		`data: {"type":"response.output_item.done","item":{"type":"function_call","id":"fc_3","call_id":"call_bash","name":"Bash","arguments":"{\"command\":\"ls\"}"}}`,
		``,
		`data: {"type":"response.completed","response":{"id":"resp-1","status":"completed","usage":{"input_tokens":120,"output_tokens":30,"input_tokens_details":{"cached_tokens":100}}}}`,
		``,
	}, "\n")
	round2 := `{"id":"resp-2","status":"completed","output":[{"type":"message","content":[{"type":"output_text","text":"Done <ok> & dusted."}]}],"usage":{"input_tokens":200,"output_tokens":5}}`
	var mu sync.Mutex
	var bodies []string
	var headers []map[string]string
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		b, _ := io.ReadAll(r.Body)
		mu.Lock()
		n := len(bodies)
		bodies = append(bodies, string(b))
		headers = append(headers, map[string]string{
			"path": r.URL.Path, "authorization": r.Header.Get("Authorization"), "accept": r.Header.Get("Accept"),
			"contentType": r.Header.Get("Content-Type"), "xExtra": r.Header.Get("X-Extra"),
		})
		mu.Unlock()
		if n == 0 {
			w.Header().Set("Content-Type", "text/event-stream")
			_, _ = w.Write([]byte(round1))
			return
		}
		w.Header().Set("Content-Type", "application/json")
		_, _ = w.Write([]byte(round2))
	}))
	defer srv.Close()
	p := testCompatibleProvider(srv.URL)
	p.ExtraHeaders = map[string]string{"X-Extra": "1", "": "dropped"}
	p.Models[0].ContextWindow = 400000
	r := &HarnessRunner{Provider: p, Client: responses.NewClient(p), Todos: &agenttools.TodoStore{}}
	inject := make(chan string, 2)
	inject <- encodeInjectPlain("Also check docs", "Also check docs (ui)")
	inject <- "   "
	spec := AgentSpec{
		Provider: p.ID, Model: "gpt-5", Flags: map[string]string{"effort": "high", "verbosity": "low"},
		Prompt: "Track the work <now>", SystemPrompt: "role", AppendSystemPrompt: "append",
		AllowedTools:    []string{"Read", "TodoWrite", "mcp__recur__ask_user", "mcp__recur__report_finding"},
		IncludePartials: true, RecurSessionID: "sess-1", WorkspaceID: "W1", AgentID: "W1-PA",
		Inject: inject, InjectPlain: true,
	}
	var events []map[string]any
	res, err := r.Run(t.Context(), spec, func(ev ParsedEvent) {
		e := map[string]any{"kind": string(ev.Kind), "text": ev.Text, "display": ev.DisplayText, "tool": ev.Tool,
			"toolUseId": ev.ToolUseID, "toolInput": ev.ToolInput, "isError": ev.IsError, "model": ev.Model, "sessionId": ev.SessionID}
		if ev.ToolInputRaw != nil {
			e["toolInputRaw"] = string(ev.ToolInputRaw)
		}
		if ev.Result != nil {
			rr := *ev.Result
			rr.DurationMS = 0
			e["result"] = rr
		}
		events = append(events, e)
	})
	if err != nil {
		t.Fatal(err)
	}
	res.DurationMS = 0
	return map[string]any{
		"round1": round1, "round2": round2, "provider": p, "bodies": bodies, "headers": headers,
		"events": events, "result": res,
	}
}
