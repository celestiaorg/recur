func dumpResult(r Result) map[string]any {
	return map[string]any{
		"sessionId": r.SessionID, "subtype": r.Subtype, "isError": r.IsError, "durationMs": r.DurationMS,
		"text": r.Text, "inputTokens": r.InputTokens, "outputTokens": r.OutputTokens,
		"cacheCreationTokens": r.CacheCreationTokens, "cacheReadTokens": r.CacheReadTokens, "stderr": r.Stderr,
	}
}

func dumpEvent(e ParsedEvent) map[string]any {
	m := map[string]any{"kind": string(e.Kind)}
	switch e.Kind {
	case KindText, KindTextDelta, KindThinking, KindThinkingDelta, KindNotice:
		m["text"] = e.Text
	case KindStep:
		m["id"], m["name"], m["summary"], m["input"] = e.ToolUseID, e.Tool, e.ToolInput, string(e.ToolInputRaw)
	case KindToolResult:
		m["id"], m["name"], m["text"], m["isError"], m["truncated"] = e.ToolUseID, e.Tool, e.Text, e.IsError, e.Truncated
	case KindUsage:
		m["result"] = dumpResult(*e.Result)
	case KindCompact:
		m["status"], m["trigger"], m["summary"], m["preTokens"] = e.CompactStatus, e.CompactTrigger, e.Text, e.Tokens
	case KindResult:
		m["result"] = dumpResult(*e.Result)
	case KindUser:
		m["text"], m["display"], m["live"] = e.Text, e.DisplayText, e.LiveSteer
	}
	return m
}
