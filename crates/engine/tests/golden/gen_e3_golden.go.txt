package engine

// Generator for crates/engine/tests/golden/go_e3.json (slice E3, R2).
// Run with RECUR_E3_GOLDEN=1 go test ./internal/engine -run TestE3GoldenDump.
// Walk ids are written as $ID, commit hashes as $HASH, durations as $DUR.

import (
	"encoding/json"
	"os"
	"regexp"
	"strings"
	"testing"
	"time"

	"recur/internal/bundles"
	"recur/internal/domain"
	"recur/internal/mcp"
)

type e3Case struct {
	In  any `json:"in"`
	Out any `json:"out"`
}

var (
	e3Hash = regexp.MustCompile(`\b[0-9a-f]{7,40}\b`)
	e3Dur  = regexp.MustCompile(`Completed in [^.·<]+`)
)

func e3Scrub(s, id string) string {
	if id != "" {
		s = strings.ReplaceAll(s, id, "$ID")
	}
	s = e3Hash.ReplaceAllString(s, "$$HASH")
	return e3Dur.ReplaceAllString(s, "Completed in $$DUR")
}

func e3Esc(esc domain.Escalation, id string) map[string]any {
	opts := []map[string]string{}
	for _, o := range esc.Options {
		opts = append(opts, map[string]string{"text": o.Text, "tag": o.Tag, "desc": o.Desc})
	}
	return map[string]any{
		"question": e3Scrub(esc.Question, id),
		"context":  e3Scrub(esc.Context, id),
		"options":  opts,
	}
}

func e3MaestroFeed(st interface{ Feed(string) []domain.FeedItem }, id string) []string {
	out := []string{}
	for _, it := range st.Feed(id) {
		if it.AgentCode != maestroAgentCode {
			continue
		}
		out = append(out, it.Name+": "+e3Scrub(it.Text, id))
	}
	return out
}

func TestE3GoldenDump(t *testing.T) {
	if os.Getenv("RECUR_E3_GOLDEN") != "1" {
		t.Skip("set RECUR_E3_GOLDEN=1 to regenerate the E3 golden file")
	}
	out := map[string]any{}
	out["constants"] = map[string]string{
		"errAlreadyDispatched":               errAlreadyDispatched.Error(),
		"errLoopOwnerGone":                   errLoopOwnerGone.Error(),
		"errVerifyRejected":                  errVerifyRejected.Error(),
		"errPhaseFailed":                     errPhaseFailed.Error(),
		"boardCapPhase":                      boardCapPhase,
		"wrapUpSystemPrompt":                 wrapUpSystemPrompt,
		"firstWorkspaceRenamePrompt":         firstWorkspaceRenamePrompt,
		"firstWorkspaceRenamePromptNoBranch": firstWorkspaceRenamePromptNoBranch,
		"firstChatRenamePrompt":              firstChatRenamePrompt,
		"rerunReviewInstruction":             "Re-review the current diff against the base branch and file findings for any real issues.",
		"maestroReplanTools":                 strings.Join(maestroReplanTools, ","),
	}

	var prTitles []e3Case
	for _, g := range []string{"", "Add dark mode\n\nwith details", strings.Repeat("é", 100), "  spaced  goal  "} {
		prTitles = append(prTitles, e3Case{g, prTitle(g)})
	}
	out["prTitle"] = prTitles

	var intros []e3Case
	for _, in := range [][2]string{{" goal ", ""}, {"recipe wire", "Review & fix"}, {"g", "  "}} {
		intros = append(intros, e3Case{in, prFallbackIntro(in[0], in[1])})
	}
	out["prFallbackIntro"] = intros

	commits := []domain.Commit{{Hash: "abc1234", Message: "Add the toggle\n\nbody"}, {Hash: "def5678", Message: "Fix lint"}}
	out["githubPRMarkdown"] = []e3Case{
		{map[string]any{"goal": "Ship dark mode", "commits": commits, "fileCount": 3, "add": 40, "del": 2, "notes": []string{" note one ", "", "note two"}},
			githubPRMarkdown("Ship dark mode", commits, 3, 40, 2, []string{" note one ", "", "note two"})},
		{map[string]any{"goal": "", "commits": []domain.Commit{}, "fileCount": 1, "add": 1, "del": 0, "notes": []string{}},
			githubPRMarkdown("", nil, 1, 1, 0, nil)},
	}

	var handoffs []e3Case
	for _, n := range []string{"", "  ", "Skipped S2; the API is missing."} {
		handoffs = append(handoffs, e3Case{n, handoffSection(n)})
	}
	out["handoffSection"] = handoffs
	out["maestroReplanSystemPrompt"] = e3Case{[]string{"Fix checkout", "1. mutex\n2. retry", "S2 split"},
		maestroReplanSystemPrompt("Fix checkout", "1. mutex\n2. retry", "S2 split")}
	out["dispatchReviewerSystemPrompt"] = e3Case{"Look at auth", dispatchReviewerSystemPrompt("Look at auth")}

	var thresholds []e3Case
	for _, th := range []bundles.FindingsThreshold{
		{AllZero: true},
		{},
		{Levels: map[string]int{"high": 0, "medium": 2}},
		{Levels: map[string]int{"low": 5, "critical": 0, "high": 1, "medium": 3}},
	} {
		thresholds = append(thresholds, e3Case{th, formatFindingsThreshold(th)})
	}
	out["formatFindingsThreshold"] = thresholds

	var wraps []e3Case
	for _, p := range []string{
		"Root cause: race in checkout.\nLanded retries in the parent and both children.",
		"  ",
		"<script>x</script> & more",
		"```\nRoot cause: nil deref\n```",
		strings.Repeat("long line words ", 120),
	} {
		wraps = append(wraps, e3Case{p, map[string]string{"html": loopWrapUpHTML(p), "sanitized": sanitizeWrapProse(p)}})
	}
	out["loopWrapUp"] = wraps

	var elapsed []e3Case
	for _, ms := range []int64{0, 400, 3000, 90_000, 5 * 60_000, 62 * 60_000, 2 * 3600_000, 1500, 59_600} {
		elapsed = append(elapsed, e3Case{ms, formatLoopElapsed(time.Duration(ms) * time.Millisecond)})
	}
	out["formatLoopElapsed"] = elapsed

	var facts []e3Case
	for _, f := range []string{"Completed in 3s.", "Completed.", "Converged.", `Completed in 1m · <a class="loop-pr-link" data-pr="1">Local PR #1</a>`} {
		facts = append(facts, e3Case{f, isCompletionFact(f)})
	}
	out["isCompletionFact"] = facts

	var tags []e3Case
	for _, s := range []string{"<p>Hello <b>world</b></p>", "a &amp; b", "no tags", "<br/>line<br>two"} {
		tags = append(tags, e3Case{s, stripTags(s)})
	}
	out["stripTags"] = tags

	var titles []e3Case
	for _, s := range []string{"add a dark-mode toggle", "  Fix   the flaky checkout test that fails on CI every other run please  ", "", "⟦attachment:img1⟧ look", strings.Repeat("word ", 30)} {
		titles = append(titles, e3Case{s, map[string]string{"provisional": provisionalWorkspaceTitle(s), "words3": truncateWords(s, 3)}})
	}
	out["titles"] = titles
	out["standingPromptSHA"] = e3Case{"You are the Maestro.", standingPromptSHA("You are the Maestro.")}

	parent := domain.Workspace{ID: "P1", Goal: "  Fix flaky checkout  ", Plan: &domain.Plan{Summary: "Two fixes", Body: "Detail body"}}
	steps := []domain.PlanStep{
		{ID: "S1", Title: "Mutex the cart", Files: []string{"cart.go", "api.go"}},
		{ID: "S3", Title: "Retry token", Files: []string{"api.go"}},
	}
	out["childStepPlan"] = e3Case{map[string]any{"parent": parent, "steps": steps}, childStepPlan(parent, steps)}

	pr := domain.PR{
		Number: 7, Title: "Dark mode",
		Comments: []domain.PRComment{
			{Kind: "description", BodyMD: "desc"},
			{Author: "Ana", BodyMD: "Looks good overall"},
			{Author: "", BodyHTML: "<p>Rename <b>this</b></p>"},
			{Author: "Bo", Path: "main.go", Line: 12, BodyMD: "nil check", Resolved: true},
			{Author: "Cy", CommitSHA: "abc", BodyMD: "commit note"},
		},
		PendingComments: []domain.PendingLineComment{{File: "a.go", Line: 3, Body: "pending one"}, {Body: "general"}},
	}
	out["formatPRThread"] = e3Case{pr, formatPRThread(pr)}

	// Walks: the cap escalation, the plan-approval escalation, and the
	// Maestro's feed lines on a converge-to-PR run.
	{
		runner := &convergeRunner{findingsPerIter: []int{2, 2, 2, 2, 2, 2, 2, 2}}
		eng, st, _, _ := newTestEngine(t, runner)
		projID := addProject(t, st, initRepo(t))
		id, err := eng.StartWorkspace(projID, "never converges, must escalate", "", "")
		if err != nil {
			t.Fatal(err)
		}
		waitFor(t, 15*time.Second, "cap escalation", func() bool {
			l, _ := st.Workspace(id)
			return l.State == domain.StateEscalated
		})
		esc, _ := st.Escalation(id)
		out["capEscalation"] = e3Esc(esc, id)
		if _, ok := eng.AnswerEscalation(id, "Stop", ""); !ok {
			t.Fatal("answer")
		}
		waitFor(t, 10*time.Second, "stopped", func() bool {
			l, _ := st.Workspace(id)
			return l.State == domain.StateStopped && eng.runningLoopByID(id) == nil
		})
		out["capFeed"] = e3MaestroFeed(st, id)
	}
	{
		runner := &captureRunner{linesFor: func(call int) [][]byte { return maestroFixture("gate", "OK.") }}
		eng, st, _, _ := newTestEngine(t, runner)
		projID := addProject(t, st, initRepo(t))
		ws, err := eng.CreateWorkspace(projID, "gated", "")
		if err != nil {
			t.Fatal(err)
		}
		_ = engineReporter{eng: eng}.SubmitPlan(ws.ID, "PA", mcp.PlanSubmission{
			Summary: "Wire the toggle", Steps: []string{"Edit settings", "Add persistence"},
		})
		if _, err := eng.StartRunOnWorkspace(ws.ID, "ship the toggle"); err != nil {
			t.Fatal(err)
		}
		waitFor(t, 5*time.Second, "plan escalation", func() bool {
			l, _ := st.Workspace(ws.ID)
			return l.State == domain.StateEscalated
		})
		esc, _ := st.Escalation(ws.ID)
		out["planEscalation"] = e3Esc(esc, ws.ID)
	}
	{
		runner := &convergeRunner{findingsPerIter: []int{1, 0}}
		eng, st, _, _ := newTestEngine(t, runner)
		projID := addProject(t, st, initRepo(t))
		id, err := eng.StartWorkspace(projID, "Add a converging feature", "", "")
		if err != nil {
			t.Fatal(err)
		}
		waitFor(t, 15*time.Second, "pr_open", func() bool {
			l, _ := st.Workspace(id)
			return l.State == domain.StatePROpen && eng.runningLoopByID(id) == nil
		})
		out["convergeFeed"] = e3MaestroFeed(st, id)
	}

	b, err := json.MarshalIndent(out, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	dst := "../../crates/engine/tests/golden/go_e3.json"
	if err := os.WriteFile(dst, append(b, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
