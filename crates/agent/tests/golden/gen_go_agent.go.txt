func TestDumpRustAgentGolden(t *testing.T) {
	path := os.Getenv("RECUR_RUST_GOLDEN")
	if path == "" {
		t.Skip()
	}
	labels := map[string]string{}
	for k, v := range recurToolLabels {
		labels[k] = v
	}
	out := map[string]any{
		"constants": map[string]any{
			"transientMarkers":       transientMarkers,
			"codexLoginExpired":      codexLoginExpiredMsg,
			"recurToolLabels":        labels,
			"permissionPromptTool":   permissionPromptTool,
			"attachmentMarker":       AttachmentMarker("ID"),
			"displayHiddenSentinel":  displayHiddenSentinel,
			"feedInputLimit":         feedInputLimit,
			"chatDetailLimit":        chatDetailLimit,
			"planMaxPersistChars":    planMaxPersistChars,
			"maxChatAttachmentBytes": maxChatAttachmentBytes,
			"steerMessage":           formatSteerUserMessage("SUMMARY"),
			"imageMime":              chatImageMIME,
		},
	}

	long := strings.Repeat("long line — with dashes ", 340)
	var summaries [][]string
	for _, c := range summaryCases(long) {
		tool, input := c[0], c[1]
		raw := json.RawMessage(input)
		summaries = append(summaries, []string{
			tool, input,
			summarizeToolInput(tool, raw),
			chatToolInput(tool, raw),
			toolDescription(tool, raw),
			permissionSummary(tool, raw),
			permissionToolLabel(tool),
		})
	}
	out["summaries"] = summaries

	var results [][]string
	for _, c := range resultCases(long) {
		results = append(results, []string{c[0], c[1], chatToolResult(c[0], c[1])})
	}
	out["results"] = results

	var trunc [][]any
	for _, s := range []string{"", "a", "hello", "a—b—c", "日本語テキスト", "😀😀😀", "x…y"} {
		for n := 0; n <= 12; n++ {
			trunc = append(trunc, []any{s, n, truncate(s, n), hardTruncateRunes(s, n)})
		}
	}
	out["truncate"] = trunc

	type failure struct {
		Err       *string `json:"err"`
		Subtype   string  `json:"subtype"`
		Text      string  `json:"text"`
		Stderr    string  `json:"stderr"`
		IsError   bool    `json:"isError"`
		Reason    string  `json:"reason"`
		Transient bool    `json:"transient"`
	}
	var failures []failure
	for _, c := range failureCases(long) {
		var err error
		if c.Err != nil {
			err = errors.New(*c.Err)
		}
		res := Result{Subtype: c.Subtype, Text: c.Text, Stderr: c.Stderr, IsError: c.IsError}
		c.Reason, c.Transient = classifyAgentError(err, res)
		if c.Reason != agentFailureReason(err, res) {
			t.Fatal("reason mismatch")
		}
		failures = append(failures, c)
	}
	out["failures"] = failures

	var tidy, first [][]string
	for _, s := range lineCases(long) {
		tidy = append(tidy, []string{s, tidyAgentError(s)})
		first = append(first, []string{s, firstLine(s)})
	}
	out["tidy"] = tidy
	out["firstLine"] = first

	type blocksCase struct {
		Text    string                `json:"text"`
		Display string                `json:"display"`
		Imgs    []string              `json:"imgs"`
		Blocks  []domain.ContentBlock `json:"blocks"`
		Order   []string              `json:"order"`
		Ensure  string                `json:"ensure"`
		Strip   string                `json:"strip"`
	}
	var blocks []blocksCase
	m := AttachmentMarker
	pre := forkPreambleLead + " Continue from the <transcript> below.\n\n<transcript>\nold ⟦attachment:other⟧\n</transcript>\n"
	for _, c := range [][]string{
		{"before " + m("a") + " after", "", "a"},
		{"W1 " + m("a") + " W2", "D1 " + m("a") + " D2", "a"},
		{"W1 " + m("a") + " W2", "composer label", "a"},
		{"W1 " + m("a") + " W2 " + m("b"), "D1 " + m("b") + " D2 " + m("a"), "a,b"},
		{pre + "please " + m("shot") + " thanks", "please " + m("shot") + " thanks", "shot"},
		{pre + m("shot"), m("shot"), "shot"},
		{pre + m("shot") + " cues", m("shot") + " cues", "shot"},
		{m("a") + m("b"), "", "b,a"},
		{"   ", "x", ""},
		{"hi", "shown", ""},
		{"x " + m("gone"), "", "a"},
		{"⟦attachment:⟧ ⟦attachment:a\nb⟧ ⟦attachment:a", "", "a\nb,a"},
		{"look\n" + m("img1") + "\nplease", "", "img1,img2"},
		{"", "", "a,b"},
		{"text\n", "", "a"},
	} {
		var imgs []UserImage
		var ids []string
		if c[2] != "" {
			ids = strings.Split(c[2], ",")
		}
		for _, id := range ids {
			imgs = append(imgs, UserImage{ID: id, Name: id + ".png", Mime: "image/png", Path: "workspaces/W/attachments/" + id + ".png"})
		}
		var order []string
		for _, img := range imagesInMarkerOrder(c[0], imgs) {
			order = append(order, img.ID)
		}
		blocks = append(blocks, blocksCase{
			Text: c[0], Display: c[1], Imgs: ids,
			Blocks: userContentBlocks(c[0], c[1], imgs),
			Order:  order,
			Ensure: ensureAttachmentMarkers(c[0], imgs),
			Strip:  stripAttachmentMarkers(c[0]),
		})
	}
	out["contentBlocks"] = blocks

	var sanitize [][]string
	for _, s := range []string{"", " a/b..c-d_e ", strings.Repeat("x", 70), "img_1-ok", "日本-1", strings.Repeat("é", 40)} {
		sanitize = append(sanitize, []string{s, sanitizeAttachmentID(s)})
	}
	out["sanitize"] = sanitize
	var ordinals []string
	for n := 0; n <= 12; n++ {
		ordinals = append(ordinals, ordinalWord(n))
	}
	out["ordinals"] = ordinals

	b, err := json.MarshalIndent(out, "", " ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(path, append(b, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}

func failureCases(long string) []struct {
	Err       *string `json:"err"`
	Subtype   string  `json:"subtype"`
	Text      string  `json:"text"`
	Stderr    string  `json:"stderr"`
	IsError   bool    `json:"isError"`
	Reason    string  `json:"reason"`
	Transient bool    `json:"transient"`
} {
	type f = struct {
		Err       *string `json:"err"`
		Subtype   string  `json:"subtype"`
		Text      string  `json:"text"`
		Stderr    string  `json:"stderr"`
		IsError   bool    `json:"isError"`
		Reason    string  `json:"reason"`
		Transient bool    `json:"transient"`
	}
	s := func(v string) *string { return &v }
	var out []f
	for _, m := range transientMarkers {
		out = append(out, f{Stderr: "Error: " + strings.ToUpper(m) + " happened"})
	}
	out = append(out,
		f{},
		f{Err: s("")},
		f{Err: s("  ")},
		f{Subtype: "error_max_turns", IsError: true},
		f{Subtype: "error_during_execution", Text: "bad thing\nsecond", IsError: true},
		f{Subtype: "success", Text: "I'll check the file"},
		f{Err: s("\x1b[31mspawn claude: ENOENT\x1b[0m"), Text: "I'll check"},
		f{Err: s("cursor bridge: [internal] internal error")},
		f{Err: s("context canceled")},
		f{Stderr: "\n  You've hit your org's monthly spend limit\nmore"},
		f{Stderr: "2026-01-01T00:00:00Z ERROR codex_login: refresh 401 {\"a\":1}"},
		f{Text: "Failed To Refresh Token"},
		f{Stderr: "2026-05-01T10:00:00.1Z ERROR codex_core::exec - stream failed {\"x\":1}"},
		f{Stderr: long},
		f{Stderr: "\x1b]0;title\x07\x1b[2K\r  warning: Rate LİMIT reached"},
		f{Err: s("merge conflict in a.go"), Subtype: "error_during_execution"},
		f{Err: s("dial tcp: lookup api: NO SUCH HOST")},
		f{Subtype: "error_x", Text: "", IsError: false},
		f{Stderr: "\t\n", Text: "t", IsError: true},
	)
	return out
}
