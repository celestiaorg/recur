func e4WantChild() []e4Case {
	yes, no := true, false
	caller := domain.Workspace{ProjectID: "p", Branch: "b", WorktreePath: "/wt"}
	type in struct {
		Caller     string `json:"caller"`
		Standalone *bool  `json:"standalone"`
		BaseBranch string `json:"baseBranch"`
		ProjectID  string `json:"projectId"`
		Mode       string `json:"mode"`
	}
	callers := map[string]domain.Workspace{
		"caller":    caller,
		"scheduled": {ProjectID: "p", Branch: "b", WorktreePath: "/wt", ScheduleID: "sched"},
		"bare":      {ProjectID: "p"},
		"nested":    {ProjectID: "p", Branch: "b", WorktreePath: "/wt", ParentID: "p0"},
	}
	out := []e4Case{}
	for _, c := range []in{
		{"caller", nil, "", "", "agent"},
		{"scheduled", nil, "", "", "agent"},
		{"caller", nil, "", "", "loop"},
		{"caller", nil, "main", "", "agent"},
		{"caller", &yes, "", "", "agent"},
		{"scheduled", &no, "", "", "agent"},
		{"caller", nil, "", "q", "agent"},
		{"bare", nil, "", "", "agent"},
		{"nested", nil, "", "", "agent"},
		{"nested", &yes, "", "", "agent"},
	} {
		w := callers[c.Caller]
		pid := c.ProjectID
		if pid == "" {
			pid = w.ProjectID
		}
		child, reason := wantChild(w, CreateWorkspaceRequest{Standalone: c.Standalone, BaseBranch: c.BaseBranch, ProjectID: c.ProjectID}, c.Mode, pid)
		out = append(out, e4Case{In: c, Out: map[string]any{"child": child, "reason": reason}})
	}
	return out
}

func TestE4GoldenDump(t *testing.T) {
	if os.Getenv("RECUR_E4_GOLDEN") == "" {

	}
	out := map[string]any{
		"mode_shapes":    e4Shapes(),
		"switch":         e4Switch(),
		"dead_ends":      e4DeadEnds(),
		"park_suffix":    switchModeParkResultSuffix,
		"peel":           e4Peel(),

		"children":       e4Children(),
		"parent_steer":   parentSteerPrefix,
		"want_child":     e4WantChild(),
		"app_chat":       e4AppChat(),
		"ages":           e4Ages(),
		"chat_nouns":     e4Nouns(),
		"briefs":         e4Briefs(),
		"fork_preamble":  e4ForkPreamble(),
		"max_remounts":   maxModeRemounts,
	}
	b, err := json.MarshalIndent(out, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile("../../crates/engine/tests/golden/go_e4.json", append(b, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
