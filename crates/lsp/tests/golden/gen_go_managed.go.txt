package managed

// Generates crates/lsp/tests/golden/go_managed.json (slice P3).
//
//	P3_MANAGED_OUT=$PWD/crates/lsp/tests/golden/go_managed.json \
//	  go test ./internal/lsp/managed -run TestP3ManagedGolden -count=1

import (
	"encoding/json"
	"os"
	"testing"
)

func TestP3ManagedGolden(t *testing.T) {
	out := os.Getenv("P3_MANAGED_OUT")
	if out == "" {
		t.Skip("P3_MANAGED_OUT not set")
	}
	asset := func(f func() (string, error)) string {
		a, err := f()
		if err != nil {
			return "error: " + err.Error()
		}
		return a
	}
	g := map[string]any{
		"releases": map[string]string{
			"gopls": GoplsRelease, "goplsModule": goplsModule, "marksman": MarksmanRelease, "ruff": RuffRelease,
			"rustAnalyzer": RustAnalyzerRelease, "taplo": TaploRelease, "typescriptLanguageServer": TypeScriptLanguageServerRelease,
			"typescript": TypeScriptRelease, "basedpyright": BasedPyrightRelease, "yaml": YAMLLanguageServerRelease,
			"bash": BashLanguageServerRelease, "svelte": SvelteLanguageServerRelease, "vue": VueLanguageServerRelease,
			"vscodeLangserversExtracted": VSCodeLangserversExtractedRelease,
		},
		"bases": map[string]string{"marksman": marksmanBase, "ruff": ruffBase, "rustAnalyzer": rustAnalyzerBase, "taplo": taploBase},
		"sha256": map[string]map[string]string{
			"marksman": marksmanSHA256, "ruff": ruffSHA256, "rustAnalyzer": rustAnalyzerSHA256, "taplo": taploSHA256,
		},
		"hostAssets": map[string]string{
			"marksman": asset(marksmanAsset), "ruff": asset(ruffAsset), "rustAnalyzer": asset(rustAnalyzerAsset), "taplo": asset(taploAsset),
		},
		"env": map[string]string{
			"gopls": EnvGopls, "rustAnalyzer": EnvRustAnalyzer, "marksman": EnvMarksman, "typescript": EnvTypeScript,
			"basedpyright": EnvBasedPyright, "ruff": EnvRuff, "taplo": EnvTaplo, "json": EnvJSON, "html": EnvHTML,
			"css": EnvCSS, "yaml": EnvYAML, "bash": EnvBash, "svelte": EnvSvelte, "vue": EnvVue,
		},
		"binDir": managedBinDir("/data", "gopls"),
	}
	raw, err := json.MarshalIndent(g, "", " ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(out, append(raw, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
