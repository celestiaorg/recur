type goldenEvent struct {
	Case    string          `json:"case"`
	Name    string          `json:"name"`
	Payload string `json:"payload"`
}

		captured = append(captured, ev)
		mu.Unlock()

		mu.Lock()

		raw, err := json.Marshal(ev.Data)

	w := ws.ID
